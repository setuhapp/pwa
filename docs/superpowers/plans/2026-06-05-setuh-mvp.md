# SETUH MVP Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the full SETUH MVP core (caregiver onboarding → tiered browse → member/bookmarks → engagement/reviews → verification/admin) as a locally-runnable Next.js + SQLite app, with payment/Aadhaar/WhatsApp/SMS integrations replaced by local substitutes.

**Architecture:** Next.js App Router + TypeScript + Tailwind, SQLite via Prisma. Mock phone-OTP cookie sessions. Six demo-able milestones, each ending with the app running, the new flow clickable against seed data, and tests green.

**Tech Stack:** Next.js (App Router), TypeScript, Tailwind CSS, Prisma + SQLite, Vitest, qrcode.react, next-pwa-style manifest.

**Reference spec:** `docs/superpowers/specs/2026-06-05-setuh-mvp-design.md`

```typescript
export type Tier = "anon" | "member" | "caregiver" | "admin";
export type SessionInfo = { userType: "caregiver" | "member" | "admin"; userId: string } | null;

// A member only gets the member tier while their subscription is active.
export function tierFromSession(session: SessionInfo, member: { subscriptionStatus: string } | null): Tier {
  if (!session) return "anon";
  if (session.userType === "admin") return "admin";
  if (session.userType === "caregiver") return "caregiver";
  if (session.userType === "member") return member?.subscriptionStatus === "active" ? "member" : "anon";
  return "anon";
}
```

- [ ] **Step 4: Implement `src/lib/session.ts` (cookie read/write + getViewer)**

```typescript
import { cookies } from "next/headers";
import { db } from "@/lib/db";
import { SESSION_COOKIE } from "@/lib/constants";
import { tierFromSession, type SessionInfo, type Tier } from "@/lib/viewer";

export async function createSession(userType: "caregiver" | "member" | "admin", userId: string) {
  const token = crypto.randomUUID();
  await db.session.create({ data: { token, userType, userId } });
  (await cookies()).set(SESSION_COOKIE, token, { httpOnly: true, sameSite: "lax", path: "/" });
}

export async function destroySession() {
  const jar = await cookies();
  const token = jar.get(SESSION_COOKIE)?.value;
  if (token) await db.session.deleteMany({ where: { token } });
  jar.delete(SESSION_COOKIE);
}

async function rawSession(): Promise<SessionInfo> {
  const token = (await cookies()).get(SESSION_COOKIE)?.value;
  if (!token) return null;
  const s = await db.session.findUnique({ where: { token } });
  return s ? { userType: s.userType as SessionInfo["userType"], userId: s.userId } : null;
}

export async function getViewer(): Promise<{ session: SessionInfo; tier: Tier }> {
  const session = await rawSession();
  let member = null;
  if (session?.userType === "member") {
    member = await db.member.findUnique({ where: { id: session.userId } });
  }
  return { session, tier: tierFromSession(session, member) };
}
```

```typescript
type CG = {
  name?: string | null; photoUrl?: string | null; address?: string | null; city?: string | null;
  skills?: string | null; qualifications?: string | null;
  experienceYears?: number | null; priorFamilies?: string | null;
  dailyRate?: number | null; monthlyRate?: number | null;
};

// Four equal sections: details, qualifications, experience, rate.
export function profileCompleteness(c: CG): number {
  const details = !!(c.name && c.photoUrl && c.address && c.city);
  const quals = !!(c.skills || c.qualifications);
  const exp = c.experienceYears != null || !!c.priorFamilies;
  const rate = c.dailyRate != null || c.monthlyRate != null;
  const filled = [details, quals, exp, rate].filter(Boolean).length;
  return filled * 25;
}
```

```typescript
import { writeFile, mkdir } from "node:fs/promises";
import path from "node:path";

// Stores uploads under public/uploads so they are served statically.
export async function saveUpload(file: File): Promise<string> {
  const bytes = Buffer.from(await file.arrayBuffer());
  const dir = path.join(process.cwd(), "public", "uploads");
  await mkdir(dir, { recursive: true });
  const name = `${crypto.randomUUID()}-${file.name.replace(/[^a-zA-Z0-9.]/g, "_")}`;
  await writeFile(path.join(dir, name), bytes);
  return `/uploads/${name}`;
}
```

```typescript
import { UNSIGNED_PROFILE_LIMIT } from "@/lib/constants";
import type { Tier } from "@/lib/viewer";

export function applyCap(orderedIds: string[], tier: Tier, alreadyViewed: string[]) {
  if (tier !== "anon") return { visibleIds: orderedIds, capped: false };
  const remaining = Math.max(0, UNSIGNED_PROFILE_LIMIT - alreadyViewed.length);
  const visibleIds = orderedIds.slice(0, remaining);
  const capped = orderedIds.length > visibleIds.length || remaining === 0;
  return { visibleIds, capped };
}
```

```typescript
import { REVIEW_UNLOCK_DAYS } from "@/lib/constants";

export type EngStatus = "pending" | "active" | "ended";
export type EngAction = "confirm" | "end";

export function reviewUnlockDate(start: Date): Date {
  return new Date(start.getTime() + REVIEW_UNLOCK_DAYS * 24 * 60 * 60 * 1000);
}

export function canReview(
  e: { status: EngStatus | string; confirmedAt: Date | null; startDate: Date },
  now: Date,
): boolean {
  if (e.status !== "active" && e.status !== "ended") return false;
  if (!e.confirmedAt) return false;
  return now.getTime() >= reviewUnlockDate(e.startDate).getTime();
}

export function nextStatus(current: EngStatus | string, action: EngAction): EngStatus | null {
  if (current === "pending" && action === "confirm") return "active";
  if (current === "active" && action === "end") return "ended";
  return null;
}
```

```typescript
  const status = nextStatus(e.status, "confirm");
  if (!status) redirect("/caregiver/confirmations");
  await db.engagement.update({ where: { id }, data: { status, confirmedAt: new Date() } });
  revalidatePath("/caregiver/confirmations");
}

export async function endEngagement(formData: FormData) {
  const memberId = await requireMember();
  const id = String(formData.get("id"));
  const e = await db.engagement.findUnique({ where: { id } });
  if (!e || e.memberId !== memberId) redirect("/member/engagements");
  const status = nextStatus(e.status, "end");
  if (!status) redirect("/member/engagements");
  await db.engagement.update({ where: { id }, data: { status, endedAt: new Date() } });
  revalidatePath("/member/engagements");
}

export async function submitReview(formData: FormData) {
  const memberId = await requireMember();
  const id = String(formData.get("engagementId"));
  const e = await db.engagement.findUnique({ where: { id } });
  if (!e || e.memberId !== memberId) redirect("/member/engagements");
  if (!canReview(e, new Date())) redirect("/member/engagements"); // server-enforced gate
  const n = (k: string) => Math.max(1, Math.min(5, Number(formData.get(k) || 0)));
  await db.review.create({
    data: {
      engagementId: id,
      reliability: n("reliability"), hygiene: n("hygiene"), gentleness: n("gentleness"),
      communication: n("communication"), overall: n("overall"),
    },
  });
  revalidatePath(`/c/${e.caregiverId}`);
  redirect(`/member/engagements`);
}
```

```typescript
const TYPE_LABEL: Record<string, string> = { aadhaar: "Aadhaar", background: "Background check", reference: "Reference" };

function fmt(d: Date): string {
  return d.toLocaleDateString("en-GB", { day: "2-digit", month: "short", year: "numeric" });
}

export function badgeLabel(v: { type: string; status: string; checkedDate: Date | null; expiryDate: Date | null }): string {
  const label = TYPE_LABEL[v.type] ?? v.type;
  if (v.status === "verified") return `${label} verified · ${v.checkedDate ? fmt(v.checkedDate) : "date unknown"}`;
  if (v.status === "self_reported") return `${label} (self-reported)`;
  return `${label} not verified`;
}

export function isVerificationCurrent(v: { status: string; expiryDate: Date | null }, now: Date): boolean {
  if (v.status !== "verified") return false;
  if (!v.expiryDate) return true;
  return v.expiryDate.getTime() > now.getTime();
}
```

```typescript
"use server";
import { db } from "@/lib/db";
import { getViewer } from "@/lib/session";
import { VERIFICATION_VALID_MONTHS } from "@/lib/constants";
import { redirect } from "next/navigation";
import { revalidatePath } from "next/cache";

async function requireAdmin() {
  const { session } = await getViewer();
  if (session?.userType !== "admin") redirect("/login?role=admin");
  return session.userId;
}

export async function addVerification(formData: FormData) {
  await requireAdmin();
  const caregiverId = String(formData.get("caregiverId"));
  const type = String(formData.get("type"));
  const status = String(formData.get("status"));
  const checkedDate = status === "verified" ? new Date() : null;
  const expiryDate = checkedDate ? new Date(new Date(checkedDate).setMonth(checkedDate.getMonth() + VERIFICATION_VALID_MONTHS)) : null;
  await db.verification.create({ data: { caregiverId, type, status, checkedDate, expiryDate, checkedBy: "SETUH admin" } });
  revalidatePath(`/admin/caregivers/${caregiverId}`);
  revalidatePath(`/c/${caregiverId}`);
}

export async function hardDeleteCaregiver(formData: FormData) {
  await requireAdmin();
  const id = String(formData.get("id"));
  await db.review.deleteMany({ where: { engagement: { caregiverId: id } } });
  await db.engagement.deleteMany({ where: { caregiverId: id } });
  await db.verification.deleteMany({ where: { caregiverId: id } });
  await db.bookmark.deleteMany({ where: { caregiverId: id } });
  await db.caregiver.delete({ where: { id } });
  revalidatePath("/admin/caregivers");
}
```
